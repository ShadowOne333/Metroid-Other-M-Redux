// Subtitles Format:
// 0C A3 XX XX YY YY 08 16 ZZ ZZ
// 0C A3	=	Display subtitle (Next 2 bytes = subtitle ID, Following 2 bytes = duration)
// XX XX	=	Subtitle ID / Message ID
// YY YY	=	Subtitle duration (in frames)
// 08 16	= 	Set frame timestamp
Opcode 0x0CA3 = 
// ZZ ZZ	=	Subtitle trigger frame (in frames)
// 0C A3 06 2B 00 9B 08 16 43 99
//------------------------------------
// dm36to43 (Ridley fight): Script indices 1664–1665 at file offsets 0x711438 and 0x711534, subtitles 1273–1297
// dm52to56 (Sector Zero/Madeline Bergman): Script indices 1751–1752 at file offsets 0x711648 and 0x711AD8, subtitles 1341–1457

// To resync subtitles, find all 08 16 XX XX 0C A3 sequences in those four blocks and change the XX XX frame values.
// Each 0816 XXXX is a 2-byte frame number (big-endian) at which the following subtitle appears.
